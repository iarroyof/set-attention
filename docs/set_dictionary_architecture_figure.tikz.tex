\node[mathbox, right=20mm of coarsestack, yshift=8mm] (eqs) {
\textbf{Set-mediated causal prediction (per group $\nu$)}\\[0.5mm]
\begin{align*}
\text{1. anchor:}\quad
  h_t^{(0)} &= e(x_t)+p_t,\\
\text{2. sets:}\quad
  S^{(\nu)}_m &= \{j:\, a^{(\nu)}_m\le j\le e^{(\nu)}_m\},\\
  \mathcal C^{(\nu)}_t
    &= \{m:\, t-w_\nu<e^{(\nu)}_m\le t\},\\
\text{3. pool:}\quad
  s^{(0,\nu)}_m
    &= \sum_{j\in S^{(\nu)}_m}\omega^{(\nu)}_{j,m}\,h^{(0)}_j,\\
\text{4. set stack:}\quad
  s^{(k+\frac12,\nu)}
    &= s^{(k,\nu)} + \mathcal A^{(\nu)}_{\le e_m}\!\bigl(\operatorname{LN}s^{(k,\nu)}\bigr),\\
  s^{(k+1,\nu)}
    &= s^{(k+\frac12,\nu)} + \mathcal F^{(\nu)}\!\bigl(\operatorname{LN}s^{(k+\frac12,\nu)}\bigr),\\
\text{5. route:}\quad
  \pi_{t,m}^{(u)}
    &= \operatorname*{softmax}_{m\in\mathcal C^{(\nu(u))}_t}
       \bigl(\widetilde g_{t,m}^{(u)}/\tau_r^{\mathrm{eff}}\bigr),\\
  r_t^{(u)}
    &= \sum_m \pi_{t,m}^{(u)}\,\widetilde s^{(u)}_m,
       \quad u\in\mathcal U_\nu,\\
\text{6. merge:}\quad
  r_t &= \operatorname{Concat}_{u=1}^{H_r} r_t^{(u)},
       \quad \operatorname{span}_t = W_o\,r_t,\\
\text{7. decode:}\quad
  f_t &= h_t^{(0)}+\operatorname{span}_t,\\
  p_{t+1} &= \operatorname{softmax}\bigl(\mathbf W_{\mathrm{lm}}\,f_t\bigr).
\end{align*}
};

\node[mathbox, below=8mm of eqs, text width=96mm] (legend) {
\textbf{SD-9 mixed-resolution head split.}
\[
\begin{gathered}
\nu\in\{\mathrm f,\mathrm c\},\qquad
(w_{\mathrm f},s_{\mathrm f})=(2,1),\qquad
(w_{\mathrm c},s_{\mathrm c})=(4,2),\\[1mm]
\mathcal U_{\mathrm f}\sqcup\mathcal U_{\mathrm c}=\{1,\dots,H_r\},\qquad
\nu(u)=\text{group of head }u.
\end{gathered}
\]
Here $\widetilde s^{(u)}_m$ is the head-$u$ subvector of the processed atom
$s^{(K,\nu(u))}_m$.  The routed span is the analogue of the transformer's
multi-head value update; the purple rail carries only the thin anchor
$h_t^{(0)}$ and is \emph{not} a contextual token tower.  In every reported run
$W_o$ is the identity, so $\operatorname{span}_t=r_t$.
};
