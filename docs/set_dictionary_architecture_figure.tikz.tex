\textbf{Aligned computation.} For $g\in\mathcal G=\{\mathrm f,\mathrm c\}$:
\[
\begin{aligned}
&\text{anchor:}       & h_t^{(0)} &= e(x_t)+p_t,\\
&\text{bank:}         & S_m^g &= \{a_m^g,\ldots,e_m^g\},\quad
                         e_m^g=ms_g+w_g-1,\\
&\text{fiber:}        & \mathcal C_t^g &= \{m:\ t-w_g<e_m^g\le t\},\\
&\text{pool:}         & z_m^{0,g} &= \mathcal P_g(\{h_j^{(0)}:j\in S_m^g\}),\\
&\text{set stack:}    & z^{K,g} &= \Phi_g(z^{0,g};\ e_j^g\le e_m^g),\\
&\text{route:}        & \pi_{t,m}^{u,g}
  &= \operatorname*{softmax}_{m\in\mathcal C_t^g}
     (\widetilde g_{t,m}^{u,g}/\tau_{r,\mathrm{eff}}),\\
&& r_t^{u,g} &= \sum_{m\in\mathcal C_t^g}
     \pi_{t,m}^{u,g}\operatorname{slice}_{u,g}(z_m^{K,g}),\\
&\text{merge:}        & r_t &= \operatorname{Concat}_{g}\operatorname{Concat}_{u\in\mathcal U_g}r_t^{u,g},\\
&\text{span:}         & \operatorname{span}_t &= W_o r_t,\\
&\text{decode:}       & f_t &= h_t^{(0)}+\operatorname{span}_t,\quad
                         \ell_t=W_{\rm lm}f_t .
\end{aligned}
\]
};
